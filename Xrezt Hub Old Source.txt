--// Movement keyword matching: single cached scan instead of 26 finds/anim
local MovementStateKeywords = {
    "run","chase","down","fall","ragdoll","idle","walk","jump","dash","climb",
    "getup","land","sprint","movement","turn","halt","hover","sleep","emote",
    "spawn","dance","wave","sit","block"
}
local movementCache = setmetatable({}, { __mode = "k" })
local function IsMovementName(name)
    local cached = movementCache[name]
    if cached ~= nil then return cached end
    local l = lower(name)
    local hit = false
    for i = 1, #MovementStateKeywords do
        if find(l, MovementStateKeywords[i], 1, true) then hit = true break end
    end
    movementCache[name] = hit
    return hit
end

--// Cache anim id -> raw digits, avoids a string.match every AnimationPlayed
local rawIdCache = {}
local function RawId(animationId)
    local c = rawIdCache[animationId]
    if c == nil then
        c = match(animationId, "%d+") or false
        rawIdCache[animationId] = c
    end
    return c or nil
end

--// Knit service cache -------------------------------------------------------
local cachedServices, remoteCache = nil, {}
local function GetKnitServices()
    if cachedServices and cachedServices.Parent then return cachedServices end
    local knit = ReplicatedStorage:FindFirstChild("Knit")
    local inner = knit and knit:FindFirstChild("Knit")
    cachedServices = inner and inner:FindFirstChild("Services") or nil
    return cachedServices
end

-- one lookup per (service, remote) pair, then cached forever
local function GetRemote(serviceName, remoteName)
    local key = serviceName .. "/" .. remoteName
    local r = remoteCache[key]
    if r and r.Parent then return r end
    local services = GetKnitServices()
    local svc = services and services:FindFirstChild(serviceName)
    local re  = svc and svc:FindFirstChild("RE")
    r = re and re:FindFirstChild(remoteName) or nil
    remoteCache[key] = r
    return r
end

local function FireRemote(serviceName, remoteName, ...)
    local remote = GetRemote(serviceName, remoteName)
    if not remote then return false end
    local ok = pcall(remote.FireServer, remote, ...)
    return ok
end

local MAHITO_BF_ANIM = "72475960800126"

    if MahitoConnection then MahitoConnection:Disconnect() MahitoConnection = nil end
    if not character then return end

    local animator = hum and hum:WaitForChild("Animator", 5)

--// Namecall hook: single early-out, no table alloc on the common path --------
local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    if getnamecallmethod() ~= "FireServer" then
        return oldNamecall(self, ...)
    end

    local name = rawget(self, "Name") or self.Name
    if name ~= "Activated" then return oldNamecall(self, ...) end

    local re = self.Parent
    local service = re and re.Parent
    if not service or re.Name ~= "RE" then return oldNamecall(self, ...) end

    -- Always Downslam: rewrite arg 1 only when it's the equipped moveset
    if AlwaysDownslamActive and service.Name == (LocalPlayer:GetAttribute("Moveset") or "") .. "Service" then
        local args = { ... }
        if args[1] == false then
            args[1] = "Down"
            return oldNamecall(self, table.unpack(args))
        end
    end

    -- BFC dash-behind
    if service.Name == "DivergentFistService"
        and not dashInProgress
        and tog("BfcEnabled") and tog("BfcDashBehind") then
        task.delay(0.08, function()
            if startCurveGlide then startCurveGlide() end
        end)
    end

    return oldNamecall(self, ...)
end)

--// Block ref-counting --------------------------------------------------------
local GlobalBlockCount = 0
local function FireBlockActivated()
    if GlobalBlockCount == 0 then FireRemote("BlockService", "Activated") end
    GlobalBlockCount += 1
end
local function FireBlockDeactivated()
    GlobalBlockCount = max(0, GlobalBlockCount - 1)
    if GlobalBlockCount == 0 then FireRemote("BlockService", "Deactivated") end
end
local function ForceReleaseBlock()
    GlobalBlockCount = 0
    FireRemote("BlockService", "Deactivated")
end

--// State checks --------------------------------------------------------------
local function IsLocalPlayerAttacking()
    if not LocalHum then return false end
    for _, track in ipairs(LocalHum:GetPlayingAnimationTracks()) do
        if track.IsPlaying and not IsMovementName(track.Animation.Name) then return true end
    end
    return false
end

local function IsIncapacitated(char)
    if not char then return true end
    if char:GetAttribute("Dead") then return true end
    local rag = char:GetAttribute("Ragdoll")
    return (rag and rag > 0) or false
end

local function RegisterTrackToDropdown(animationName)
    if knownAnimNames[animationName] or IsMovementName(animationName) then return end
    knownAnimNames[animationName] = true
    if AllAnimationNames[1] == "Loading animations..." then tremove(AllAnimationNames, 1) end
    tinsert(AllAnimationNames, animationName)
    dropdownDirty = true
end

end)

--// Feint detection (cached per emitter) --------------------------------------
local feintCache = setmetatable({}, { __mode = "k" })
local function IsEnemyFeinting(character)
    if not tog("AntiFeintEnabled") then return false end
    local root = character:FindFirstChild("HumanoidRootPart")
    local feint = root and root:FindFirstChild("Feint")
    local ring = feint and feint:FindFirstChild("Ring")
    if not (ring and ring:IsA("ParticleEmitter")) then return false end

    local cached = feintCache[ring]
    if cached ~= nil then return cached end
    local grey = false
    for _, kp in ipairs(ring.Color.Keypoints) do
        local v = kp.Value
        if abs(v.R - v.G) < 0.05 and abs(v.G - v.B) < 0.05 then grey = true break end
    end
    feintCache[ring] = grey
    return grey
end

    if not LocalChar then return nil end

    if attr then return attr end
    local obj = LocalChar:FindFirstChild("Moveset")
    if not obj then return nil end
    return obj:IsA("StringValue") and obj.Value or obj.Name
end

--// Shared release helper (kills the duplicated Stopped/timeout blocks) --------
local function ReleaseOn(tracker, track, timeout)
    local conn
    conn = track.Stopped:Once(function()
        if tracker[track] then tracker[track] = nil FireBlockDeactivated() end
    end)
    tinsert(ActiveConnections, conn)
    task.delay(timeout or 5, function()
        if tracker[track] then tracker[track] = nil FireBlockDeactivated() end
    end)
end

--// Projectile block ----------------------------------------------------------
local function HandleProjectileAnimationPlay(character, track)
    if not tog("ProjectileBlock") then return end
    local anim = track and track.Animation
    if not anim then return end

    local rawId = RawId(anim.AnimationId)
    if tog("DebugMode") then
        print("[XREZT] Projectile raw:", rawId, "inList:", rawId and ProjectileAnimations[rawId] or false)
    end
    if not rawId or not ProjectileAnimations[rawId] then return end
    if Players:GetPlayerFromCharacter(character) == LocalPlayer then return end
    if IsIncapacitated(LocalChar) or not LocalRoot then return end

    local targetRoot = character:FindFirstChild("HumanoidRootPart")
    if not targetRoot then return end

    local delta = targetRoot.Position - LocalRoot.Position
    local range = opt("ProjectileRange", 100)
    if delta.Magnitude > range then return end

    if tog("ProjectileFacingCheck") then
        local dot = targetRoot.CFrame.LookVector:Dot(-delta.Unit)
        if dot < opt("ProjectileFacingThreshold", 0.5) then return end
    end

    if ActiveProjectileTracks[track] then return end
    ActiveProjectileTracks[track] = true
    FireBlockActivated()
    ReleaseOn(ActiveProjectileTracks, track, 5)
end

--// Standard M1/attack block ---------------------------------------------------
local function HandleStandardAnimationPlay(character, track)
    if not tog("AutoBlockEnabled") then return end
    local anim = track and track.Animation
    if not anim then return end

    local rawId = RawId(anim.AnimationId)
    if rawId and ProjectileAnimations[rawId] then return end

    local animName = anim.Name
    if IsMovementName(animName) then return end
    if ExclusionSet[animName] then return end
    if Players:GetPlayerFromCharacter(character) == LocalPlayer then return end
    if not tog("BlockWhileAttacking") and IsLocalPlayerAttacking() then return end
    if IsIncapacitated(LocalChar) or IsIncapacitated(character) then return end
    if not LocalRoot then return end

    local targetRoot = character:FindFirstChild("HumanoidRootPart")
    if not targetRoot then return end

    local delta = targetRoot.Position - LocalRoot.Position
    if delta.Magnitude > opt("DetectionRange", 25) then return end

    if tog("FacingCheckEnabled") then
        local dot = targetRoot.CFrame.LookVector:Dot(-delta.Unit)
        if dot < opt("FacingAngleThreshold", 0.5) then return end
    end
    if IsEnemyFeinting(character) then return end
    if ActiveBlockingTracks[track] then return end

    ActiveBlockingTracks[track] = true
    FireBlockActivated()

    if tog("BlockForAnimation") then
        ReleaseOn(ActiveBlockingTracks, track, 5)
    elseif tog("BlockForDuration") then
        task.delay(opt("BlockDuration", 0.5), function()
            if ActiveBlockingTracks[track] then
                ActiveBlockingTracks[track] = nil
                FireBlockDeactivated()
            end
        end)
    end
end
