-- Memoized keyword test: the same animation names repeat constantly,
-- so we only pay for the linear scan once per unique name.
local MovementNameCache = {}
local function IsMovementName(animationName)
    local cached = MovementNameCache[animationName]
    if cached ~= nil then return cached end
    local lowerName = string_lower(animationName)
    local result = false
    for i = 1, MOVEMENT_KEYWORD_COUNT do
        if string_find(lowerName, MovementStateKeywords[i], 1, true) then
            result = true
            break
        end
    end
    MovementNameCache[animationName] = result
    return result
end

local function SetupMahitoCharacter(character)
    if MahitoConnection then
        MahitoConnection:Disconnect()
        MahitoConnection = nil
    end
    if not character then return end
    local humanoid = character:WaitForChild("Humanoid", 5)
    if not humanoid then return end
    local animator = humanoid:WaitForChild("Animator", 5)
    if not animator then return end

    MahitoConnection = animator.AnimationPlayed:Connect(function(track)
        local toggle = Toggles.AutoMahitoBlackflash
        if not toggle or not toggle.Value then return end
        local anim = track.Animation
        if not anim then return end
        if not string_find(anim.AnimationId, MAHITO_BF_ANIM, 1, true) then return end

        task.spawn(function()
            task.wait(0.2)
            pcall(function()
                local services = GetKnitServices()
                local fss = services and services:FindFirstChild("FocusStrikeService")
                local re = fss and fss:FindFirstChild("RE")
                local activated = re and re:FindFirstChild("Activated")
                local moveset = character:FindFirstChild("Moveset")
                local focusStrike = moveset and moveset:FindFirstChild("Focus Strike")
                if activated and focusStrike then
                    activated:FireServer(focusStrike)
                end
            end)
        end)
    end)
    table.insert(ActiveConnections, MahitoConnection)
end

--// Namecall hook: fast-path out before allocating the varargs table.
local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    if method ~= "FireServer" then
        return oldNamecall(self, ...)
    end

    local bfcToggle = Toggles.BfcEnabled
    local bfcOn = bfcToggle and bfcToggle.Value
    if not AlwaysDownslamActive and not bfcOn then
        return oldNamecall(self, ...)
    end

    local parent = self.Parent
    local isActivatedRemote = self.Name == "Activated" and parent and parent.Name == "RE"

    if AlwaysDownslamActive and isActivatedRemote then
        local service = parent.Parent
        local moveset = LocalPlayer:GetAttribute("Moveset")
        if service and moveset and service.Name == (moveset .. "Service") then
            local args = { ... }
            if args[1] == false then
                args[1] = "Down"
            end
            return oldNamecall(self, table_unpack(args))
        end
    end

    if bfcOn and not dashInProgress and isActivatedRemote then
        local service = parent.Parent
        if service and service.Name == "DivergentFistService" then
            local dashBehind = Toggles.BfcDashBehind
            if dashBehind and dashBehind.Value then
                task.spawn(function()
                    task.wait(0.08)
                    startCurveGlide()
                end)
            end
        end
    end

    return oldNamecall(self, ...)
end)

--// Block reference counting
local GlobalBlockCount = 0

local function GetBlockRemote(name)
    local services = GetKnitServices()
    local blockService = services and services:FindFirstChild("BlockService")
    local re = blockService and blockService:FindFirstChild("RE")
    return re and re:FindFirstChild(name)
end

local function FireBlockActivated()
    if GlobalBlockCount == 0 then
        local activated = GetBlockRemote("Activated")
        if activated then
            pcall(function() activated:FireServer() end)
        end
    end
    GlobalBlockCount = GlobalBlockCount + 1
end

local function FireBlockDeactivated()
    GlobalBlockCount = math_max(0, GlobalBlockCount - 1)
    if GlobalBlockCount == 0 then
        local deactivated = GetBlockRemote("Deactivated")
        if deactivated then
            pcall(function() deactivated:FireServer() end)
        end
    end
end

local function ForceReleaseBlock()
    GlobalBlockCount = 0
    local deactivated = GetBlockRemote("Deactivated")
    if deactivated then
        pcall(function() deactivated:FireServer() end)
    end
end

--// Local player state helpers
local function IsLocalPlayerAttacking()
    local localChar = LocalPlayer.Character
    if not localChar then return false end
    local humanoid = localChar:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    for _, track in ipairs(humanoid:GetPlayingAnimationTracks()) do
        if track.IsPlaying then
            local anim = track.Animation
            if anim and not IsMovementName(anim.Name) then
                return true
            end
        end
    end
    return false
end

local function IsLocalPlayerIncapacitated()
    local localChar = LocalPlayer.Character
    if not localChar then return true end
    if localChar:GetAttribute("Dead") then return true end
    local ragdollVal = localChar:GetAttribute("Ragdoll")
    if ragdollVal and ragdollVal > 0 then return true end
    return false
end

local function RegisterTrackToDropdown(animationName)
    if IsMovementName(animationName) then return end
    if table.find(AllAnimationNames, animationName) then return end
    if AllAnimationNames[1] == "Loading animations..." then
        table.remove(AllAnimationNames, 1)
    end
    table.insert(AllAnimationNames, animationName)
    table.sort(AllAnimationNames)
    if Options.AnimationsList then
        Options.AnimationsList:SetValues(AllAnimationNames)
    end
end

local function IsEnemyFeinting(character)
    local toggle = Toggles.AntiFeintEnabled
    if not toggle or not toggle.Value then return false end
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return false end
    local feintInstance = rootPart:FindFirstChild("Feint")
    if not feintInstance then return false end
    local ringParticle = feintInstance:FindFirstChild("Ring")
    if not (ringParticle and ringParticle:IsA("ParticleEmitter")) then return false end
    for _, keypoint in ipairs(ringParticle.Color.Keypoints) do
        local val = keypoint.Value
        if math_abs(val.R - val.G) < 0.05 and math_abs(val.G - val.B) < 0.05 then
            return true
        end
    end
    return false
end

local function IsProjectileFacingUs(targetRoot, localRoot)
    local directionToUs = (localRoot.Position - targetRoot.Position).Unit
    return targetRoot.CFrame.LookVector:Dot(directionToUs) >= Options.ProjectileFacingThreshold.Value
end

--// Projectile auto-block
local function HandleProjectileAnimationPlay(character, animationTrack)
    if not Toggles.ProjectileBlock.Value then return end
    if not animationTrack then return end
    local anim = animationTrack.Animation
    if not anim then return end

    local rawId = string_match(anim.AnimationId, "%d+")
    if Toggles.DebugMode.Value then
        print("[XREZT DEBUG] Projectile - Raw: " .. tostring(rawId) ..
              " InList: " .. tostring(rawId and ProjectileAnimations[rawId] or false))
    end
    if not rawId or not ProjectileAnimations[rawId] then return end
    if ActiveProjectileTracks[animationTrack] then return end
    if Players:GetPlayerFromCharacter(character) == LocalPlayer then return end
    if IsLocalPlayerIncapacitated() then return end

    local localChar = LocalPlayer.Character
    if not localChar then return end
    local localRoot = localChar:FindFirstChild("HumanoidRootPart")
    local targetRoot = character:FindFirstChild("HumanoidRootPart")
    if not localRoot or not targetRoot then return end
    if (localRoot.Position - targetRoot.Position).Magnitude > Options.ProjectileRange.Value then return end
    if Toggles.ProjectileFacingCheck.Value and not IsProjectileFacingUs(targetRoot, localRoot) then return end

    ActiveProjectileTracks[animationTrack] = true
    FireBlockActivated()

    -- :Once self-disconnects, so it must NOT be appended to ActiveConnections.
    animationTrack.Stopped:Once(function()
        if ActiveProjectileTracks[animationTrack] then
            ActiveProjectileTracks[animationTrack] = nil
            FireBlockDeactivated()
        end
    end)

    task.delay(5, function()
        if ActiveProjectileTracks[animationTrack] then
            ActiveProjectileTracks[animationTrack] = nil
            FireBlockDeactivated()
        end
    end)
end

--// Standard auto-block (cheap checks first, attacking check last)
local function HandleStandardAnimationPlay(character, animationTrack)
    if not Toggles.AutoBlockEnabled.Value then return end
    if not animationTrack then return end
    local anim = animationTrack.Animation
    if not anim then return end
    if ActiveBlockingTracks[animationTrack] then return end

    local animationName = anim.Name
    local rawId = string_match(anim.AnimationId, "%d+")
    if rawId and ProjectileAnimations[rawId] then return end
    if IsMovementName(animationName) then return end
    if table.find(AnimationExclusionList, animationName) then return end
    if Players:GetPlayerFromCharacter(character) == LocalPlayer then return end
    if character:GetAttribute("Dead") then return end

    local ragdollVal = character:GetAttribute("Ragdoll")
    if ragdollVal and ragdollVal > 0 then return end
    if IsLocalPlayerIncapacitated() then return end

    local localChar = LocalPlayer.Character
    if not localChar then return end
    local localRoot = localChar:FindFirstChild("HumanoidRootPart")
    local targetRoot = character:FindFirstChild("HumanoidRootPart")
    if not localRoot or not targetRoot then return end
    if (localRoot.Position - targetRoot.Position).Magnitude > Options.DetectionRange.Value then return end

    if Toggles.FacingCheckEnabled.Value then
        local directionToUs = (localRoot.Position - targetRoot.Position).Unit
        if targetRoot.CFrame.LookVector:Dot(directionToUs) < Options.FacingAngleThreshold.Value then return end
    end

    if IsEnemyFeinting(character) then return end
    -- Most expensive check (allocates a track table) runs last.
    if not Toggles.BlockWhileAttacking.Value and IsLocalPlayerAttacking() then return end

    ActiveBlockingTracks[animationTrack] = true
    FireBlockActivated()

    if Toggles.BlockForAnimation.Value then
        animationTrack.Stopped:Once(function()
            if ActiveBlockingTracks[animationTrack] then
                ActiveBlockingTracks[animationTrack] = nil
                FireBlockDeactivated()
            end
        end)
        task.delay(5, function()
            if ActiveBlockingTracks[animationTrack] then
                ActiveBlockingTracks[animationTrack] = nil
                FireBlockDeactivated()
            end
        end)
    elseif Toggles.BlockForDuration.Value then
        task.delay(Options.BlockDuration.Value, function()
            if ActiveBlockingTracks[animationTrack] then
                ActiveBlockingTracks[animationTrack] = nil
                FireBlockDeactivated()
            end
        end)
    end
end

--// BFC curve glide (assigned, not re-declared, so the hook above sees it)
function startCurveGlide()
    local mode = Options.BfcDashMode and Options.BfcDashMode.Value or "Blatant"
    if mode == "Legit" then
        if startLegitCurveGlide then startLegitCurveGlide() end
        return
    end
    if not getNearestTarget then return end
    local target = getNearestTarget()
    if not target then
