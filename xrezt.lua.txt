local GlobalBlockCount = 0

local function FireBlockActivated()
    if GlobalBlockCount == 0 then
        local services = GetKnitServices()
        local blockService = services and services:FindFirstChild("BlockService")
        local re = blockService and blockService:FindFirstChild("RE")
        local activated = re and re:FindFirstChild("Activated")
        if activated then
            pcall(function() activated:FireServer() end)
        end
    end
    GlobalBlockCount = GlobalBlockCount + 1
end

local function FireBlockDeactivated()
    GlobalBlockCount = math_max(0, GlobalBlockCount - 1)
    if GlobalBlockCount == 0 then
        local services = GetKnitServices()
        local blockService = services and services:FindFirstChild("BlockService")
        local re = blockService and blockService:FindFirstChild("RE")
        local deactivated = re and re:FindFirstChild("Deactivated")
        if deactivated then
            pcall(function() deactivated:FireServer() end)
        end
    end
end

local function IsLocalPlayerAttacking()
    local localChar = LocalPlayer.Character
    if not localChar then return false end
    local humanoid = localChar:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    for _, track in pairs(humanoid:GetPlayingAnimationTracks()) do
        if track.IsPlaying then
            local animName = string.lower(track.Animation.Name)
            local isMovement = false
            for i = 1, #MovementStateKeywords do
                if string.find(animName, MovementStateKeywords[i]) then 
                    isMovement = true 
                    break 
                end
            end
            if not isMovement then return true end
        end
    end
    return false
end

local function IsLocalPlayerIncapacitated()
    local localChar = LocalPlayer.Character
    if not localChar then return true end
    if localChar:GetAttribute("Dead") then return true end
    local ragdollVal = localChar:GetAttribute("Ragdoll")
    if ragdollVal and ragdollVal > 0 then return true end
    return false
end

local function RegisterTrackToDropdown(animationName)
    local lowerName = string.lower(animationName)
    for i = 1, #MovementStateKeywords do
        if string.find(lowerName, MovementStateKeywords[i]) then return end
    end
    if not table.find(AllAnimationNames, animationName) then
        if AllAnimationNames[1] == "Loading animations..." then
            table.remove(AllAnimationNames, 1)
        end
        table.insert(AllAnimationNames, animationName)
        table.sort(AllAnimationNames)
        if Options.AnimationsList then Options.AnimationsList:SetValues(AllAnimationNames) end
    end
end

local function IsEnemyFeinting(character)
    if not Toggles.AntiFeintEnabled or not Toggles.AntiFeintEnabled.Value then return false end
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return false end
    local feintInstance = rootPart:FindFirstChild("Feint")
    if feintInstance then
        local ringParticle = feintInstance:FindFirstChild("Ring")
        if ringParticle and ringParticle:IsA("ParticleEmitter") then
            for _, keypoint in ipairs(ringParticle.Color.Keypoints) do
                local val = keypoint.Value
                if math_abs(val.R - val.G) < 0.05 and math_abs(val.G - val.B) < 0.05 then
                    return true
                end
            end
        end
    end
    return false
end

local function IsProjectileFacingUs(targetRoot, localRoot)
    local enemyLookVector = targetRoot.CFrame.LookVector
    local directionToUs = (localRoot.Position - targetRoot.Position).Unit
    local dotProduct = enemyLookVector:Dot(directionToUs)
    return dotProduct >= Options.ProjectileFacingThreshold.Value
end

local function HandleProjectileAnimationPlay(character, animationTrack)
    if not Toggles.ProjectileBlock.Value then return end
    if not animationTrack or not animationTrack.Animation then return end
    local animIdStr = tostring(animationTrack.Animation.AnimationId)
    local rawId = string.match(animIdStr, "%d+")
    if Toggles.DebugMode.Value then
        print("[XREZT DEBUG] Projectile - Raw: " .. tostring(rawId) .. " InList: " .. tostring(rawId and ProjectileAnimations[rawId] or false))
    end
    if not rawId or not ProjectileAnimations[rawId] then return end
    local charPlayer = Players:GetPlayerFromCharacter(character)
    if charPlayer == LocalPlayer then return end
    if IsLocalPlayerIncapacitated() then return end
    local localChar = LocalPlayer.Character
    if not localChar then return end
    local localRoot = localChar:FindFirstChild("HumanoidRootPart")
    local targetRoot = character:FindFirstChild("HumanoidRootPart")
    if not localRoot or not targetRoot then return end
    local distance = (localRoot.Position - targetRoot.Position).Magnitude
    if distance > Options.ProjectileRange.Value then return end
    if Toggles.ProjectileFacingCheck.Value and not IsProjectileFacingUs(targetRoot, localRoot) then return end
    if ActiveProjectileTracks[animationTrack] then return end
    ActiveProjectileTracks[animationTrack] = true
    FireBlockActivated()
    local connection
    connection = animationTrack.Stopped:Once(function()
        if ActiveProjectileTracks[animationTrack] then
            ActiveProjectileTracks[animationTrack] = nil
            FireBlockDeactivated()
        end
    end)
    table.insert(ActiveConnections, connection)
    task.delay(5, function()
        if ActiveProjectileTracks[animationTrack] then
            ActiveProjectileTracks[animationTrack] = nil
            FireBlockDeactivated()
        end
    end)
end

local function HandleStandardAnimationPlay(character, animationTrack)
    if not Toggles.AutoBlockEnabled.Value then return end
    if not animationTrack or not animationTrack.Animation then return end
    local animationName = animationTrack.Animation.Name
    local lowerName = string.lower(animationName)
    local rawId = string.match(animationTrack.Animation.AnimationId, "%d+")
    if rawId and ProjectileAnimations[rawId] then return end
    for i = 1, #MovementStateKeywords do
        if string.find(lowerName, MovementStateKeywords[i]) then return end
    end
    local charPlayer = Players:GetPlayerFromCharacter(character)
    if charPlayer == LocalPlayer then return end
    if not Toggles.BlockWhileAttacking.Value and IsLocalPlayerAttacking() then return end
    if IsLocalPlayerIncapacitated() then return end
    if character:GetAttribute("Dead") then return end
    local ragdollVal = character:GetAttribute("Ragdoll")
    if ragdollVal and ragdollVal > 0 then return end
    local localChar = LocalPlayer.Character
    if not localChar then return end
    local localRoot = localChar:FindFirstChild("HumanoidRootPart")
    local targetRoot = character:FindFirstChild("HumanoidRootPart")
    if not localRoot or not targetRoot then return end
    local distance = (localRoot.Position - targetRoot.Position).Magnitude
    if distance > Options.DetectionRange.Value then return end
    if Toggles.FacingCheckEnabled.Value then
        local enemyLookVector = targetRoot.CFrame.LookVector
        local directionToUs = (localRoot.Position - targetRoot.Position).Unit
        local facingDotProduct = enemyLookVector:Dot(directionToUs)
        if facingDotProduct < Options.FacingAngleThreshold.Value then return end
    end
    if IsEnemyFeinting(character) then return end
    if table.find(AnimationExclusionList, animationName) then return end
    if ActiveBlockingTracks[animationTrack] then return end
    ActiveBlockingTracks[animationTrack] = true
    FireBlockActivated()
    if Toggles.BlockForAnimation.Value then
        local connection
        connection = animationTrack.Stopped:Once(function()
            if ActiveBlockingTracks[animationTrack] then
                ActiveBlockingTracks[animationTrack] = nil
                FireBlockDeactivated()
            end
        end)
        table.insert(ActiveConnections, connection)
        task.delay(5, function()
            if ActiveBlockingTracks[animationTrack] then
                ActiveBlockingTracks[animationTrack] = nil
                FireBlockDeactivated()
            end
        end)
    elseif Toggles.BlockForDuration.Value then
        task.spawn(function()
            task.wait(Options.BlockDuration.Value)
            if ActiveBlockingTracks[animationTrack] then
                ActiveBlockingTracks[animationTrack] = nil
                FireBlockDeactivated()
            end
        end)
    end
end

local qtePollActive = false
local qteHeartbeatConnection = nil

local function fireQTEConnections(button)
    if not button then return end
    if getconnections then
        for _, eventName in ipairs({"MouseButton1Down", "MouseButton1Click", "Activated"}) do
            local event = button[eventName]
            if event then
                for _, connection in ipairs(getconnections(event)) do
                    pcall(function()
                        connection:Fire()
                        connection:Fire()
                        connection:Fire()
                    end)
                end
            end
        end
    else
        pcall(function()
            button:Activate()
        end)
    end
end

local function processAutoQTE()
    local playerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")
    if not playerGui then return end
    
    local qteInstance = playerGui:FindFirstChild("QTE")
    if qteInstance then
        for burst = 1, 10 do
            if UserInputService.TouchEnabled then
                local mobileBtn = qteInstance:FindFirstChild("QTE_MOBILE")
                if mobileBtn then
                    fireQTEConnections(mobileBtn)
                end
            else
                local pcLabel = qteInstance:FindFirstChild("QTE_PC")
                if pcLabel and pcLabel:IsA("TextLabel") then
                    local keyStr = pcLabel.Text
                    if keyStr and keyStr ~= "" then
                        local success, keyCode = pcall(function() return Enum.KeyCode[keyStr] end)
                        if success and keyCode then
                            VirtualInputManager:SendKeyEvent(true, keyCode, false, game)
                            VirtualInputManager:SendKeyEvent(false, keyCode, false, game)
                        end
                    end
                end
            end
        end
    end
end

local function startAutoQTE()
    qtePollActive = true
    
    task.spawn(function()
        while qtePollActive do
            pcall(processAutoQTE)
            task.wait()
        end
    end)
    
    qteHeartbeatConnection = RunService.Heartbeat:Connect(function()
        pcall(processAutoQTE)
    end)
end

local function stopAutoQTE()
    qtePollActive = false
    if qteHeartbeatConnection then
        qteHeartbeatConnection:Disconnect()
        qteHeartbeatConnection = nil
    end
end

local function GetAimlockTarget()

    if not localRoot then return nil end

    local currentCamera = workspace.CurrentCamera
    local screenCenter = Vector2_new(currentCamera.ViewportSize.X / 2, currentCamera.ViewportSize.Y / 2)

    if Toggles.HardLockEnabled and Toggles.HardLockEnabled.Value and AimlockTarget then
        local targetRoot = AimlockTarget:FindFirstChild("HumanoidRootPart")
        local targetHumanoid = AimlockTarget:FindFirstChildOfClass("Humanoid")
        if targetRoot and targetHumanoid and targetHumanoid.Health > 0 and not AimlockTarget:GetAttribute("Dead") and AimlockTarget.Parent then
            local distance = (localRoot.Position - targetRoot.Position).Magnitude
            if distance <= Options.AimlockRange.Value then

            end
        end
        AimlockTarget = nil
    end

    local closestTarget = nil

    local camPos = currentCamera.CFrame.Position

    local function evaluateCharacter(char)
        if char == localChar then return end
        local targetRoot = char:FindFirstChild("HumanoidRootPart")
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if not targetRoot or not humanoid or humanoid.Health <= 0 then return end
        if char:GetAttribute("Dead") then return end

        local distance = (localRoot.Position - targetRoot.Position).Magnitude
        if distance > Options.AimlockRange.Value then return end

        staticRayParams.FilterDescendantsInstances = {localChar}
        local result = workspace:Raycast(camPos, targetRoot.Position - camPos, staticRayParams)
        if result and not result.Instance:IsDescendantOf(char) then return end

        local screenPos, onScreen = currentCamera:WorldToViewportPoint(targetRoot.Position)
        if not onScreen then return end

        local screenOffset = (screenCenter - Vector2_new(screenPos.X, screenPos.Y)).Magnitude
        -- Weight screen distance heavily with localized distances to preserve flawless lock stability
        local score = screenOffset * 3.5 + distance * 0.15

        if score < bestScore then
            bestScore = score
            closestTarget = char
        end
    end

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            evaluateCharacter(plr.Character)
        end
    end

    local CharactersFolder = Workspace:FindFirstChild("Characters")
    if CharactersFolder then
        for _, char in pairs(CharactersFolder:GetChildren()) do

            end
        end
    end

            rawVelocity = (targetPos - AimlockLastTargetPos) / (now - AimlockLastTargetTime)
            -- Filter anomalous speed spikes (teleports / flings) to prevent aimlock throwing errors
            if rawVelocity.Magnitude > 300 then
                rawVelocity = Vector3_new(0, 0, 0)
            end

        -- Filter raw velocity utilizing exponential moving average for ultra-smooth tracking
        local velSmoothAlpha = 1 - math_exp(-18 * dt)

            AimlockSmoothPos = predictedPosition
        else
            -- If positions desync heavily, fast snap to prevent target sliding. Else, interpolate silky-smooth.
            if (predictedPosition - AimlockSmoothPos).Magnitude > 40 then
                AimlockSmoothPos = predictedPosition
            else
                local smoothAlpha = 1 - math_exp(-15 * dt)
                AimlockSmoothPos = AimlockSmoothPos:Lerp(predictedPosition, smoothAlpha)
            end

                AimlockStoredStates.RotationType = UserGameSettings.RotationType
            end

            UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
            UserGameSettings.RotationType = Enum.RotationType.CameraRelative

            local lookAt = AimlockSmoothPos + Vector3_new(0, 1.25, 0)

                if h then h.AutoRotate = false end
            end)

            local currentPos = localRoot.Position
            local targetDir = Vector3_new(AimlockSmoothPos.X - currentPos.X, 0, AimlockSmoothPos.Z - currentPos.Z)

            if targetDir.Magnitude > 0.001 then
                targetDir = targetDir.Unit

-- Calculated from 8 projected points of character's 3D bounding box context to yield perfectly stabilized 2D bounds
local function CalculateRigid2DBounds(char)
    local rootPart = char:FindFirstChild("HumanoidRootPart")
    if not rootPart then return nil end
    
    local extentsSize = char:GetExtentsSize()
    local cf, size = char:GetBoundingBox()
    
    -- Stabilize orientation axes to prevent bounding-box warping when target runs/rotates
    local verticalY = Vector3_new(0, size.Y / 2, 0)
    local lateralX = Vector3_new(size.X / 2, 0, 0)
    local lateralZ = Vector3_new(0, 0, size.Z / 2)
    
    local rootPosition = cf.Position
    local camera = workspace.CurrentCamera
    
    local rawCorners = {
        rootPosition + lateralX + verticalY + lateralZ,
        rootPosition + lateralX + verticalY - lateralZ,
        rootPosition - lateralX + verticalY + lateralZ,
        rootPosition - lateralX + verticalY - lateralZ,
        rootPosition + lateralX - verticalY + lateralZ,
        rootPosition + lateralX - verticalY - lateralZ,
        rootPosition - lateralX - verticalY + lateralZ,
        rootPosition - lateralX - verticalY - lateralZ
    }

    local minX, minY = math.huge, math.huge
    local maxX, maxY = -math.huge, -math.huge
    local anyVisible = false

    for i = 1, 8 do
        local screenPos, onScreen = camera:WorldToViewportPoint(rawCorners[i])
        if screenPos.Z > 0 then
            anyVisible = true
            if screenPos.X < minX then minX = screenPos.X end
            if screenPos.Y < minY then minY = screenPos.Y end
            if screenPos.X > maxX then maxX = screenPos.X end
            if screenPos.Y > maxY then maxY = screenPos.Y end
        end
    end

    if not anyVisible then return nil end
    return {
        TopLeft = Vector2_new(minX, minY),
        BottomRight = Vector2_new(maxX, maxY),
        Width = maxX - minX,
        Height = maxY - minY
    }
end

local function UpdateCharEsp(char)
    local data = EspDrawings[char]
    if not data then return end
    
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    local rootPart = char:FindFirstChild("HumanoidRootPart")
    
    if not humanoid or not rootPart or humanoid.Health <= 0 or char:GetAttribute("Dead") then
        for _, drawing in pairs(data) do
            drawing.Visible = false
        end
        return
    end
    
    local bounds = CalculateRigid2DBounds(char)
    if not bounds then
        for _, drawing in pairs(data) do drawing.Visible = false end
        return
    end
    
    local camera = workspace.CurrentCamera
    local viewportSize = camera.ViewportSize
    
    local healthRatio = humanoid.Health / humanoid.MaxHealth
    local isDummy = (Players:GetPlayerFromCharacter(char) == nil)
    local useHealthColor = Toggles.HealthBasedEsp and Toggles.HealthBasedEsp.Value
    
    local boxColor
    if isDummy and Toggles.DummyEspEnabled and Toggles.DummyEspEnabled.Value then
        boxColor = Options.DummyChamsColor and Options.DummyChamsColor.Value or Color3.fromRGB(255, 165, 0)
    elseif useHealthColor then
        boxColor = GetHealthColor(healthRatio)
    else
        boxColor = Options.BoxEspColor and Options.BoxEspColor.Value or Color3.fromRGB(255, 255, 255)
    end
    
    -- Stabilized and smoothed line configurations
    if Toggles.BoxEspEnabled and Toggles.BoxEspEnabled.Value or (isDummy and Toggles.DummyEspEnabled and Toggles.DummyEspEnabled.Value) or useHealthColor then
        data.boxTop.From = bounds.TopLeft
        data.boxTop.To = Vector2_new(bounds.BottomRight.X, bounds.TopLeft.Y)
        data.boxTop.Color = boxColor
        data.boxTop.Visible = true
        
        data.boxBottom.From = Vector2_new(bounds.TopLeft.X, bounds.BottomRight.Y)
        data.boxBottom.To = bounds.BottomRight
        data.boxBottom.Color = boxColor
        data.boxBottom.Visible = true
        
        data.boxLeft.From = bounds.TopLeft
        data.boxLeft.To = Vector2_new(bounds.TopLeft.X, bounds.BottomRight.Y)
        data.boxLeft.Color = boxColor
        data.boxLeft.Visible = true
        
        data.boxRight.From = Vector2_new(bounds.BottomRight.X, bounds.TopLeft.Y)
        data.boxRight.To = bounds.BottomRight
        data.boxRight.Color = boxColor
        data.boxRight.Visible = true
    else
        data.boxTop.Visible = false
        data.boxBottom.Visible = false
        data.boxLeft.Visible = false
        data.boxRight.Visible = false
    end
    
    -- Fixed bounding misalignment offsets on healthbars
    if Toggles.BoxEspEnabled and Toggles.BoxEspEnabled.Value and not isDummy then
        local barWidth = 4
        local barGap = 4
        local barHeight = bounds.Height
        local barX = bounds.TopLeft.X - barWidth - barGap
        local barY = bounds.TopLeft.Y
        
        data.healthBg.Size = Vector2_new(barWidth, barHeight)
        data.healthBg.Position = Vector2_new(barX, barY)
        data.healthBg.Visible = true
        
        local fillHeight = math_max(1, barHeight * healthRatio)
        data.healthFill.Size = Vector2_new(barWidth, fillHeight)
        data.healthFill.Position = Vector2_new(barX, barY + (barHeight - fillHeight))
        data.healthFill.Color = GetHealthColor(healthRatio)
        data.healthFill.Visible = true
    else
        data.healthBg.Visible = false
        data.healthFill.Visible = false
    end
    
    if Toggles.TracerEnabled and Toggles.TracerEnabled.Value then
        local origin = Vector2_new(viewportSize.X / 2, viewportSize.Y)
        -- Connect tracer to stabilized dynamic bottom center of character box bounds
        local targetPos = Vector2_new((bounds.TopLeft.X + bounds.BottomRight.X) / 2, bounds.BottomRight.Y)
        data.tracer.From = origin
        data.tracer.To = targetPos
        data.tracer.Color = boxColor
        data.tracer.Thickness = Options.TracerThickness and Options.TracerThickness.Value or 1
        data.tracer.Visible = true
    else
        data.tracer.Visible = false
    end
    
    if Toggles.NameEspEnabled and Toggles.NameEspEnabled.Value then
        local player = Players:GetPlayerFromCharacter(char)
        local displayName = player and player.DisplayName or (isDummy and "Dummy" or char.Name)
        local dist = (camera.CFrame.Position - rootPart.Position).Magnitude
        data.nameLabel.Text = displayName .. " [" .. math_floor(dist) .. "m]"

        data.nameLabel.Size = Options.NameTextSize and Options.NameTextSize.Value or 13
        data.nameLabel.Color = boxColor
        data.nameLabel.Visible = true
    else
        data.nameLabel.Visible = false
    end

local function RefreshItemESP()
    local itemsFolder = workspace:FindFirstChild("Items")
    if not itemsFolder then return end
    
    for item, drawing in pairs(ItemDrawings) do
        if not item or not item.Parent then
            pcall(function() drawing:Remove() end)
            ItemDrawings[item] = nil
        end
    end
    
    if not Toggles.ItemEspEnabled or not Toggles.ItemEspEnabled.Value then
        for _, drawing in pairs(ItemDrawings) do
            pcall(function() drawing:Remove() end)
        end
        table.clear(ItemDrawings)
        return
    end
    
    local localChar = LocalPlayer.Character
    local rootPart = localChar and localChar:FindFirstChild("HumanoidRootPart")
    if not rootPart then return end
    
    for _, item in ipairs(itemsFolder:GetChildren()) do
        if item:IsA("BasePart") or item:IsA("MeshPart") or item:IsA("Model") then
            local position = item:IsA("Model") and item:FindFirstChild("HumanoidRootPart") and item.HumanoidRootPart.Position or (item:IsA("BasePart") and item.Position)
            if not position then continue end
            
            local distance = (rootPart.Position - position).Magnitude
            local screenPos, onScreen = workspace.CurrentCamera:WorldToViewportPoint(position)
            if not onScreen or screenPos.Z < 0 then
                if ItemDrawings[item] then ItemDrawings[item].Visible = false end
                continue
            end
            
            if not ItemDrawings[item] then
                local text = Drawing.new("Text")
                text.Size = 12
                text.Center = true
                text.Outline = true
                text.OutlineColor = Color3.new(0, 0, 0)
                text.Font = 2
                text.Color = Color3.fromRGB(255, 255, 255)
                ItemDrawings[item] = text
            end
            
            local text = ItemDrawings[item]
            text.Text = item.Name .. " [" .. math_floor(distance) .. "m]"
            text.Position = Vector2_new(screenPos.X, screenPos.Y - 20)
            text.Visible = true
        end
    end
end

local function StartEspLoop()
    RunService:BindToRenderStep("NewEspLoop", Enum.RenderPriority.Camera.Value + 2, function()
        for char, _ in pairs(EspDrawings) do
            if not char or not char.Parent then
                DestroyCharDrawings(char)
                if DummyHighlights[char] then
                    pcall(function() DummyHighlights[char]:Destroy() end)
                    DummyHighlights[char] = nil
                end
            else
                UpdateCharEsp(char)
                UpdateDummyChams(char)
            end
        end
        
        local charsFolder = workspace:FindFirstChild("Characters")
        if charsFolder then
            for _, char in ipairs(charsFolder:GetChildren()) do
                if char:IsA("Model") and not EspDrawings[char] then
                    local player = Players:GetPlayerFromCharacter(char)
                    if player ~= LocalPlayer then
                        CreateCharDrawings(char)
                    end
                end
            end
        end
        
        if not _G.lastItemEspTime or tick() - _G.lastItemEspTime > 0.2 then
            RefreshItemESP()
            _G.lastItemEspTime = tick()
        end
    end)
end

local function CleanupEsp()
    for char, _ in pairs(EspDrawings) do
        DestroyCharDrawings(char)
    end
    for _, hl in pairs(DummyHighlights) do
        pcall(function() hl:Destroy() end)
    end
    for _, drawing in pairs(ItemDrawings) do
        pcall(function() drawing:Remove() end)
    end
    table.clear(EspDrawings)
    table.clear(DummyHighlights)
    table.clear(ItemDrawings)
    pcall(function() RunService:UnbindFromRenderStep("NewEspLoop") end)
end

local function destroyGhost()
    if ghostModel then ghostModel:Destroy() ghostModel = nil end
    positionHistory = {}
    ghostPartsCache = {}
end

local function createGhost(character)
    if ghostModel then ghostModel:Destroy() ghostModel = nil end
    if not character then return end
    ghostModel = Instance.new("Model")
    ghostModel.Name = "PingGhost"
    ghostPartsCache = {}
    local color = Options.BfcGhostColor and Options.BfcGhostColor.Value or Color3.fromRGB(255, 80, 80)
    for _, part in pairs(character:GetDescendants()) do
        if part:IsA("BasePart") or part:IsA("MeshPart") then
            local clone = part:Clone()
            for _, child in pairs(clone:GetChildren()) do
                if not child:IsA("SpecialMesh") and not child:IsA("Decal") then child:Destroy() end
            end
            clone.Anchored     = true
            clone.CanCollide   = false
            clone.CastShadow   = false
            clone.Transparency = 0.6
            clone.Color        = color
            clone.Material     = Enum.Material.Neon
            clone.Parent       = ghostModel
            ghostPartsCache[part.Name] = clone
        end
    end
    ghostModel.Parent = workspace
end

local function recordSnapshot(character)
    if not character then return end
    local snapshot = { time = tick(), parts = {} }
    for _, part in pairs(character:GetDescendants()) do
        if part:IsA("BasePart") or part:IsA("MeshPart") then
            snapshot.parts[part.Name] = part.CFrame
        end
    end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if hrp then snapshot.parts["HumanoidRootPart"] = hrp.CFrame end
    table.insert(positionHistory, snapshot)
    while #positionHistory > MAX_HISTORY do table.remove(positionHistory, 1) end
end

local function getDelayedSnapshot()
    local targetTime = tick() - networkPing
    for i = #positionHistory, 1, -1 do
        if positionHistory[i].time <= targetTime then return positionHistory[i] end
    end
    return positionHistory[1]
end

local function applySnapshotToGhost(snapshot)
    if not ghostModel or not snapshot then return end
    for name, cf in pairs(snapshot.parts) do
        local cachedPart = ghostPartsCache[name]
        if cachedPart then
            cachedPart.CFrame = cf
        end
    end
end

-- ─── Helpers ─────────────────────────────────────────────────
local function getHRP(character)
    return character and (
        character:FindFirstChild("HumanoidRootPart") or
        character:FindFirstChild("Torso") or
        character:FindFirstChild("UpperTorso")
    )
end

local function getNearestTarget()
    local myHRP = getHRP(LocalPlayer.Character)
    if not myHRP then return nil end
    local nearest, nearestDist = nil, Options.BfcTargetRange.Value

    local function consider(charModel)
        if charModel == LocalPlayer.Character then return end
        local hum = charModel:FindFirstChildOfClass("Humanoid")
        local hrp = getHRP(charModel)
        if hum and hum.Health > 0 and hrp then
            local dist = (myHRP.Position - hrp.Position).Magnitude
            if dist < nearestDist then
                nearestDist = dist
                nearest = charModel
            end
        end
    end

    for _, pl in pairs(Players:GetPlayers()) do
        if pl ~= LocalPlayer and pl.Character then
            consider(pl.Character)
        end
    end

    local charFolder = workspace:FindFirstChild("Characters")
    if charFolder then
        for _, obj in ipairs(charFolder:GetChildren()) do
            if obj:IsA("Model") then consider(obj) end
        end
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid")
           and not Players:GetPlayerFromCharacter(obj) then
            consider(obj)
        end
    end

    return nearest
end

local function isInFrontOfTarget(myHRP, tHRP)
    if not (myHRP and tHRP) then return false end
    local targetLook = Vector3_new(tHRP.CFrame.LookVector.X, 0, tHRP.CFrame.LookVector.Z)
    if targetLook.Magnitude < 0.01 then return false end
    targetLook = targetLook.Unit
    local toPlayer = Vector3_new(
        myHRP.Position.X - tHRP.Position.X,
        0,
        myHRP.Position.Z - tHRP.Position.Z
    )
    if toPlayer.Magnitude < 0.1 then return true end
    return targetLook:Dot(toPlayer.Unit) > 0
end

-- ─── Anchor Lock-On (camera + character aim locked to target center) ──
local lockOnStartTime = 0

local function stopLockOn()
    lockOnActive = false
    lockedTarget = nil
    pcall(function() RunService:UnbindFromRenderStep(LOCKON_RENDER_NAME) end)
end

local function startLockOn(targetChar)
    if not targetChar then return end
    stopLockOn()

    lockedTarget = targetChar
    lockOnActive = true
    lockOnStartTime = tick()

    RunService:BindToRenderStep(LOCKON_RENDER_NAME, Enum.RenderPriority.Camera.Value + 1, function()
        if not lockOnActive then
            stopLockOn()
            return
        end

        local elapsed = tick() - lockOnStartTime
        if elapsed >= Options.BfcLockOnDuration.Value then
            stopLockOn()
            return
        end

        -- Validate or reacquire target
        local tHRP = getHRP(lockedTarget)
        local tHum = lockedTarget and lockedTarget:FindFirstChildOfClass("Humanoid")
        if not tHRP or not tHRP.Parent or not tHum or tHum.Health <= 0 then
            local newTarget = getNearestTarget()
            if newTarget then
                lockedTarget = newTarget
                tHRP = getHRP(newTarget)
            else
                stopLockOn()
                return
            end
        end

        local myChar = LocalPlayer.Character
        local myHRP  = getHRP(myChar)

        if not (myHRP and tHRP and tHRP.Parent) then return end

        -- ─── Target center mass (torso, not feet) ────────────
        local targetCenter = tHRP.Position + Vector3_new(0, 1.8, 0)

        -- ─── Camera: behind player, aimed at target center ───
        -- Direction from target to player (flattened)
        local flatDir = Vector3_new(
            myHRP.Position.X - tHRP.Position.X,
            0,
            myHRP.Position.Z - tHRP.Position.Z
        )
        if flatDir.Magnitude < 0.1 then flatDir = Vector3_new(0, 0, 1) end
        flatDir = flatDir.Unit

        -- Camera sits behind player relative to target, slightly elevated
        local camDist   = 8.5
        local camHeight = Options.BfcCamLift.Value
        local desiredCamPos = myHRP.Position + flatDir * camDist + Vector3_new(0, camHeight, 0)

        -- Smooth lerp so the camera doesn't snap
        local currentCamPos = Camera.CFrame.Position
        local smoothedCamPos = currentCamPos:Lerp(desiredCamPos, 0.25)

        -- Aim camera at target center mass
        Camera.CFrame = CFrame.new(smoothedCamPos, targetCenter)

        -- ─── Character faces target center (flat XZ) ─────────
        -- This ensures attacks go toward the target, not the frontside
        local currentPos = myHRP.Position
        local lookAtFlat = Vector3_new(targetCenter.X, currentPos.Y, targetCenter.Z)
        myHRP.CFrame = CFrame.new(currentPos, lookAtFlat)
    end)
end

-- ─── Circular Arc Sidedash ───────────────────────────────────
local function performSidedash(targetChar, onComplete)
    local myChar = LocalPlayer.Character
    local myHRP  = getHRP(myChar)
    local tHRP   = getHRP(targetChar)

    if not (myHRP and tHRP and tHRP.Parent) then
        if onComplete then onComplete() end
        return
    end

    local hum = myChar:FindFirstChildOfClass("Humanoid")
    if not hum then
        if onComplete then onComplete() end
        return
    end

    local dashDuration = Options.BfcBlatantDuration and Options.BfcBlatantDuration.Value or 0.20

    local startPos   = myHRP.Position
    local targetPos  = tHRP.Position

    local distToTarget = (Vector3_new(startPos.X, 0, startPos.Z)
                        - Vector3_new(targetPos.X, 0, targetPos.Z)).Magnitude

    local arcRadius = math.max(distToTarget, MIN_ARC_RADIUS)

    local startAngle = math.atan2(
        startPos.Z - targetPos.Z,
        startPos.X - targetPos.X
    )

    local targetLookFlat = Vector3_new(tHRP.CFrame.LookVector.X, 0, tHRP.CFrame.LookVector.Z)
    if targetLookFlat.Magnitude < 0.01 then targetLookFlat = Vector3_new(0, 0, 1) end
    targetLookFlat = targetLookFlat.Unit

    local targetLookAngle = math.atan2(targetLookFlat.Z, targetLookFlat.X)
    local behindAngle = targetLookAngle + math.pi

    local angleDiff = behindAngle - startAngle
    while angleDiff > math.pi do angleDiff = angleDiff - 2 * math.pi end
    while angleDiff < -math.pi do angleDiff = angleDiff + 2 * math.pi end
    local endAngle = startAngle + angleDiff

    local behindPos = Vector3_new(
        targetPos.X - targetLookFlat.X * BEHIND_RADIUS,
        startPos.Y,
        targetPos.Z - targetLookFlat.Z * BEHIND_RADIUS
    )

    dashInProgress = true
    hum.AutoRotate = false

    local dashStart = tick()
    local conn

    conn = RunService.Heartbeat:Connect(function()
        local elapsed = tick() - dashStart
        local alpha = math.clamp(elapsed / dashDuration, 0, 1)

        local currentHRP  = getHRP(LocalPlayer.Character)
        local currentTHRP = getHRP(targetChar)

        if not currentHRP or not currentTHRP or not currentTHRP.Parent then
            if conn then conn:Disconnect() end
            dashInProgress = false
            if hum and hum.Parent then hum.AutoRotate = true end
            if onComplete then onComplete() end
            return
        end

        if alpha >= 1 then
            currentHRP.CFrame = CFrame.new(
                behindPos,
                Vector3_new(currentTHRP.Position.X, behindPos.Y, currentTHRP.Position.Z)
            )
            if conn then conn:Disconnect() end
            dashInProgress = false
            if hum and hum.Parent then hum.AutoRotate = true end
            if onComplete then onComplete() end
            return
        end

        local easedAlpha = alpha < 0.5
            and 2 * alpha * alpha
            or 1 - ((-2 * alpha + 2) ^ 2) / 2

        local currentAngle = startAngle + (endAngle - startAngle) * easedAlpha
        local currentRadius = arcRadius + (BEHIND_RADIUS - arcRadius) * easedAlpha

        local pos = Vector3_new(
            currentTHRP.Position.X + math.cos(currentAngle) * currentRadius,
            startPos.Y,
            currentTHRP.Position.Z + math.sin(currentAngle) * currentRadius
        )

        currentHRP.CFrame = CFrame.new(
            pos,
            Vector3_new(currentTHRP.Position.X, pos.Y, currentTHRP.Position.Z)
        )
    end)
end

-- ─── Legit Curve Glide ───────────────────────────────────────
local function startLegitCurveGlide()
    if legitGlideInProgress then return end

    local stopDist = Options.BfcStopDistance and Options.BfcStopDistance.Value or 2.5
    local duration = Options.BfcLegitDuration and Options.BfcLegitDuration.Value or 0.14
    local arcWidth = Options.BfcArcWidth     and Options.BfcArcWidth.Value     or 9
    local camLift  = Options.BfcCamLift      and Options.BfcCamLift.Value      or 4

    local target = getNearestTarget()
    local myChar = LocalPlayer.Character
    local myHRP  = getHRP(myChar)
    local hum    = myChar and myChar:FindFirstChildOfClass("Humanoid")
    if not (target and myHRP and hum) then return end

    local tHRP = getHRP(target)
    if not tHRP or not tHRP.Parent then return end

    legitGlideInProgress = true

    local startTime       = tick()
    local startPos        = Vector3_new(myHRP.Position.X, 0, myHRP.Position.Z)
    local initialRightVec = Vector3_new(tHRP.CFrame.RightVector.X, 0, tHRP.CFrame.RightVector.Z).Unit
    local toPlayer        = startPos - Vector3_new(tHRP.Position.X, 0, tHRP.Position.Z)
    local lockedSideSign  = toPlayer:Dot(initialRightVec) >= 0 and 1 or -1

    local function getBehindPos()
        if not (tHRP and tHRP.Parent) then return nil end
        local cf        = tHRP.CFrame
        local vel       = tHRP.AssemblyLinearVelocity
        local pingComp  = networkPing > 0.25 and 1.5 or (networkPing > 0.15 and 1.25 or 1.0)
        local predicted = tHRP.Position + vel * (networkPing * pingComp)
        local lookFlat  = Vector3_new(cf.LookVector.X, 0, cf.LookVector.Z).Unit
        return Vector3_new(predicted.X, 0, predicted.Z) - lookFlat * stopDist
    end

    local function getSideControlPoint()
        if not (tHRP and tHRP.Parent) then return nil end
        local rightVec = Vector3_new(tHRP.CFrame.RightVector.X, 0, tHRP.CFrame.RightVector.Z).Unit
        return Vector3_new(tHRP.Position.X, 0, tHRP.Position.Z) + rightVec * (arcWidth * lockedSideSign)
    end

    hum.AutoRotate = false
    local prevCam     = Camera.CameraType
    Camera.CameraType = Enum.CameraType.Custom

    local MAX_DEST_SPEED = 22
    local rawInit        = getBehindPos()
    local smoothedEnd    = rawInit or Vector3_new(myHRP.Position.X, 0, myHRP.Position.Z)
    local lastHB         = tick()
    local GLIDE_TIMEOUT  = duration + 1.2
    local conn

    conn = RunService.Heartbeat:Connect(function()
        local now = tick()
        local dt  = math.min(now - lastHB, 0.05)
        lastHB    = now

        local function abort()
            conn:Disconnect()
            if hum and hum.Parent then hum.AutoRotate = true end
            Camera.CameraType  = prevCam
            legitGlideInProgress = false
            lastLegitGlideEnd  = tick()
        end

        if now - startTime > GLIDE_TIMEOUT then abort() return end
        if not (Toggles.BfcEnabled and Toggles.BfcEnabled.Value) then abort() return end

        myHRP = getHRP(LocalPlayer.Character)
        if not (myHRP and myHRP.Parent and hum and hum.Health > 0) then abort() return end
        if not (tHRP and tHRP.Parent) then abort() return end

        local alpha = math_clamp((now - startTime) / duration, 0, 1)

        if alpha >= 1 then
            conn:Disconnect()
            hum.AutoRotate    = true
            Camera.CameraType = prevCam
            local finalPos = getBehindPos()
            if finalPos then
                myHRP.CFrame = CFrame_new(
                    Vector3_new(finalPos.X, myHRP.Position.Y, finalPos.Z),
                    Vector3_new(tHRP.Position.X, myHRP.Position.Y, tHRP.Position.Z)
                )
            end
            legitGlideInProgress = false
            lastLegitGlideEnd  = tick()
            task.spawn(function()
                if not (Toggles.BfcEnabled and Toggles.BfcEnabled.Value) then return end
                local delay = Options.BfcBlackFlashDelay and Options.BfcBlackFlashDelay.Value or 0.19
                task.wait(delay)
                if not (Toggles.BfcEnabled and Toggles.BfcEnabled.Value) then return end
                local char = LocalPlayer.Character
                if not char then return end
                local moveset = char:FindFirstChild("Moveset")
                if not moveset then return end
                local move = moveset:FindFirstChild("Divergent Fist")
                if not move then return end
                local re = getActivatedRE()
                if not re then return end
                re:FireServer(move)
                if target then startLockOn(target) end
            end)
            return
        end

        local rawEnd = getBehindPos()
        if not rawEnd then abort() return end
        local delta   = rawEnd - smoothedEnd
        local maxStep = MAX_DEST_SPEED * dt
        smoothedEnd   = delta.Magnitude > maxStep and smoothedEnd + delta.Unit * maxStep or rawEnd

        local controlPos = getSideControlPoint()
        if not controlPos then abort() return end

        local blendedControl = Vector3_new(
            controlPos.X * (1 - alpha) + smoothedEnd.X * alpha,
            0,
            controlPos.Z * (1 - alpha) + smoothedEnd.Z * alpha
        )

        local t  = 1 - (1 - alpha)^2
        local t1 = 1 - t
        local movePos = Vector3_new(
            (t1*t1)*startPos.X + (2*t1*t)*blendedControl.X + (t*t)*smoothedEnd.X,
            myHRP.Position.Y,
            (t1*t1)*startPos.Z + (2*t1*t)*blendedControl.Z + (t*t)*smoothedEnd.Z
        )

        myHRP.CFrame = CFrame_new(
            movePos,
            Vector3_new(tHRP.Position.X, movePos.Y, tHRP.Position.Z)
        )
        Camera.CFrame = CFrame_new(
            myHRP.Position + Vector3_new(0, camLift, 0),
            tHRP.Position
        )
    end)
end

-- ─── Manual Trigger ──────────────────────────────────────────
local function manualTrigger()
    local mode = Options.BfcDashMode and Options.BfcDashMode.Value or "Blatant"
    if not Toggles.BfcEnabled.Value then return end
    if mode == "Blatant" and dashInProgress then return end
    if mode == "Legit"   and legitGlideInProgress then return end

    local target = getNearestTarget()
    local myHRP  = getHRP(LocalPlayer.Character)

    if Toggles.BfcDashBehind.Value and target and myHRP then
        local tHRP = getHRP(target)
        if tHRP and isInFrontOfTarget(myHRP, tHRP) then
            if mode == "Legit" then
                startLegitCurveGlide()
            else
                performSidedash(target, function()
                    if Toggles.BfcEnabled.Value then
                        fireActivated()
                        startLockOn(target)
                    end
                end)
            end
            return
        end
    end

    fireActivated()
    if target then startLockOn(target) end
end

-- ─── Auto-Trigger ────────────────────────────────────────────
local function setupBfcCharacter(character)
    local humanoid = character:WaitForChild("Humanoid", 5)
    if not humanoid then return end
    local animator = humanoid:WaitForChild("Animator", 5)
    if not animator then return end

    local ANIM_COOLDOWN = 0.6
    local lastAnimFire  = 0

    animator.AnimationPlayed:Connect(function(track)
        if not Toggles.BfcEnabled.Value then return end
        local animId     = track.Animation.AnimationId
        local delayTime  = AnimationTriggers[animId]
        local isStraight = StraightAnimations[animId]
        if delayTime or isStraight then
            local now = tick()
            if now - lastAnimFire < ANIM_COOLDOWN then return end
            lastAnimFire = now

            local target = getNearestTarget()
            local myHRP  = getHRP(LocalPlayer.Character)

            local mode = Options.BfcDashMode and Options.BfcDashMode.Value or "Blatant"
            if Toggles.BfcDashBehind.Value and target and myHRP then
                local tHRP = getHRP(target)
                if tHRP and isInFrontOfTarget(myHRP, tHRP) then
                    if mode == "Legit" and not legitGlideInProgress then
                        startLegitCurveGlide()
                    elseif mode == "Blatant" and not dashInProgress then
                        performSidedash(target, function()
                            if Toggles.BfcEnabled.Value then
                                fireActivated()
                                startLockOn(target)
                            end
                        end)
                    end
                    return
                end
            end

            task.spawn(function()
                task.wait(delayTime)
                if not (humanoid.Health > 0 and Toggles.BfcEnabled.Value) then return end
                fireActivated()
                if target then startLockOn(target) end
            end)
        end
    end)
end

            ExtraConnections.PerfectSwap:Disconnect()
            ExtraConnections.PerfectSwap = nil
        end
    end
end

-- Refined Logic: Auto Ratio
local function HandleAutoRatio(state)
    local KnitServices = GetKnitServices()
    if state then
        local NanamiService = KnitServices and KnitServices:WaitForChild("NanamiService", 5)
        if NanamiService and NanamiService:FindFirstChild("RE") then
            ExtraConnections.AutoRatio = NanamiService.RE.Effects.OnClientEvent:Connect(function(...)
                local args = {...}
                if args[1] == "SpawnRatio" and args[2] == LocalPlayer then
                    local dataPing = Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
                    local timeMult = type(args[6]) == "number" and args[6] or 1
                    
                    local waitTime = timeMult * math.clamp(0.56 - ((math.floor(dataPing) - 100) / 1000) * 0.9, 0.25, 0.85)
                    task.wait(waitTime)
                    
                    NanamiService.RE.RightActivated:FireServer()
                end
            end)
        end
    else
        if ExtraConnections.AutoRatio then
            ExtraConnections.AutoRatio:Disconnect()
            ExtraConnections.AutoRatio = nil
        end
    end
end

                local function _getRandomPlayer()
                    local valid = {}
                    for _, p in ipairs(_Players:GetPlayers()) do
                        if p ~= _LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                            table.insert(valid, p)
                        end
                    end
                    if #valid == 0 then return nil end
                    return valid[math.random(1, #valid)]
                end

                local function _getCFrame()
                    local char = _LocalPlayer.Character
                    if char and char:FindFirstChild("HumanoidRootPart") then
                        return char.HumanoidRootPart.CFrame
                    end
                    return nil
                end

                local function _teleportTo(targetPlayer)
                    local char = _LocalPlayer.Character
                    local target = targetPlayer.Character
                    if not char or not target then return false end
                    local hrp = char:FindFirstChild("HumanoidRootPart")
                    local thrp = target:FindFirstChild("HumanoidRootPart")
                    if not hrp or not thrp then return false end
                    hrp.CFrame = thrp.CFrame * CFrame.new(0, 0, 4)
                    return true
                end

                local function _isRubberBanded(originalCFrame)
                    local current = _getCFrame()
                    if not current then return true end
                    return (current.Position - originalCFrame.Position).Magnitude < 8
                end

                local function _knitReset()
                    if _ResetRemote then
                        pcall(function() _ResetRemote:FireServer() end)
                    end
                end

                local function _waitForRespawn()
                    local oldChar = _LocalPlayer.Character
                    local t = 0
                    while _LocalPlayer.Character == oldChar and t < 10 do
                        task.wait(0.1)
                        t += 0.1
                    end
                    local newChar = _LocalPlayer.Character
                    if not newChar or not newChar.Parent or newChar == oldChar then
                        newChar = _LocalPlayer.CharacterAdded:Wait()
                    end
                    newChar:WaitForChild("HumanoidRootPart", 10)
                    task.wait(0.3)
                end

                local _acBypassToggle
                local _rapidTPToggle
                local _acBypassRunning = false

                local function _startACBypass()
                    if _acBypassRunning then return end
                    _acBypassRunning = true
                    local attempts = 0
                    task.spawn(function()
                        while _acBypassRunning do
                            if not _acBypassToggle.get() then
                                _acBypassRunning = false
                                break
                            end
                            local originalCFrame = _getCFrame()
                            if not originalCFrame then
                                task.wait(1)
                                continue
                            end
                            local target = _getRandomPlayer()
                            if not target then
                                _updateStatus("No valid player found", _Theme.Red)
                                task.wait(2)
                                continue
                            end
                            attempts += 1
                            _updateStatus("Attempt " .. attempts .. " → " .. target.Name, _Theme.Orange)
                            if _teleportTo(target) then
                                task.wait(0.35)
                                if _isRubberBanded(originalCFrame) then
                                    _updateStatus("Rubber-banded - resetting...", _Theme.Orange)
                                    _knitReset()
                                    _waitForRespawn()
                                    task.wait(0.3)
                                    if not _acBypassToggle.get() then
                                        _acBypassRunning = false
                                        break
                                    end
                                else
                                    _updateStatus("AC Bypassed", _Theme.Green)
                                    _TweenService:Create(_scStroke, TweenInfo.new(0.2, Enum.EasingStyle.Sine), {Color = _Theme.Green, Transparency = 0}):Play()
                                    task.delay(1.5, function()
                                        _TweenService:Create(_scStroke, TweenInfo.new(0.5, Enum.EasingStyle.Sine), {Color = _Theme.CardStroke, Transparency = 0.4}):Play()
                                    end)
                                    task.wait(0.5)
                                    local char = _LocalPlayer.Character
                                    if char and char:FindFirstChild("HumanoidRootPart") then
                                        char.HumanoidRootPart.CFrame = originalCFrame
                                    end
                                    _acBypassToggle.set(false)
                                    _acBypassRunning = false
                                    break
                                end
                            else
                                task.wait(0.5)
                            end
                        end
                    end)
                end

                local _rapidTPRunning = false
                local function _startRapidTP()
                    if _rapidTPRunning then return end
                    _rapidTPRunning = true
                    task.spawn(function()
                        while _rapidTPRunning do
                            if not _rapidTPToggle.get() then
                                _rapidTPRunning = false
                                break
                            end
                            for _, p in ipairs(_Players:GetPlayers()) do
                                if not _rapidTPRunning or not _rapidTPToggle.get() then break end
                                if p ~= _LocalPlayer then
                                    local char = _LocalPlayer.Character
                                    local target = p.Character
                                    if char and target then
                                        local hrp = char:FindFirstChild("HumanoidRootPart")
                                        local thrp = target:FindFirstChild("HumanoidRootPart")
                                        if hrp and thrp then
                                            hrp.CFrame = thrp.CFrame * CFrame.new(0, 0, 3)
                                        end
                                    end
                                    task.wait(0.05)
                                end
                            end
                            task.wait(0.1)
                        end
                    end)
                end

                        task.wait(0.08)
                    end
                end)

                    local t = 0
                    while _ScreenGui.Parent do
                        t = t + 0.04
                        local alpha = (math.sin(t) + 1) / 2
                        _g1s.Transparency = 0.55 + (alpha * 0.3)
                        _g2s.Transparency = 0.72 + (alpha * 0.25)
                        _mfStroke.Transparency = 0.2 + (alpha * 0.25)
                        task.wait(0.04)
                    end
                end)

                _updateStatus("Idle", _Theme.SubText)
            end)
        else
            local coreGui = game:GetService("CoreGui")
            local existingGui = coreGui:FindFirstChild("XreztACBypass")
            if existingGui then
                existingGui:Destroy()
            else
                local pGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")
                if pGui then
                    local g = pGui:FindFirstChild("XreztACBypass")
                    if g then g:Destroy() end
                end
            end
        end
    end

task.spawn(function()
    while true do
        task.wait(0.1)
        if not Toggles.AutoBlockEnabled.Value and not Toggles.ProjectileBlock.Value then
            HudBlockState:SetText('Block Status: Inactive'); HudClosestEnemy:SetText('Closest Threat: None'); HudEnemyDistance:SetText('Threat Distance: N/A'); continue
        end
        local closestOpponent = nil; local closestDistance = math.huge
        local localChar = LocalPlayer.Character; if not localChar then continue end
        local localRoot = localChar:FindFirstChild('HumanoidRootPart'); if not localRoot then continue end
        local CharactersFolder = Workspace:FindFirstChild('Characters')
        if CharactersFolder then
            for _, char in pairs(CharactersFolder:GetChildren()) do
                local charPlayer = Players:GetPlayerFromCharacter(char)
                if charPlayer ~= LocalPlayer and char:FindFirstChild('HumanoidRootPart') then
                    local humanoid = char:FindFirstChildOfClass('Humanoid')
                    if humanoid and humanoid.Health > 0 and not char:GetAttribute('Dead') then
                        local distance = (localRoot.Position - char.HumanoidRootPart.Position).Magnitude
                        local maxRange = math_max(Options.DetectionRange.Value, Options.ProjectileRange.Value)
                        if distance < closestDistance and distance <= maxRange then closestDistance = distance; closestOpponent = char.Name end
                    end
                end
            end
        end
        if GlobalBlockCount > 0 then HudBlockState:SetText('Block Status: <font color="#49ff49">Active</font>') else HudBlockState:SetText('Block Status: <font color="#ff3c3c">Inactive</font>') end
        if closestOpponent then HudClosestEnemy:SetText('Closest Threat: ' .. closestOpponent); HudEnemyDistance:SetText('Threat Distance: ' .. math_floor(closestDistance) .. ' studs') else HudClosestEnemy:SetText('Closest Threat: None in Range'); HudEnemyDistance:SetText('Threat Distance: N/A') end
    end
end)
